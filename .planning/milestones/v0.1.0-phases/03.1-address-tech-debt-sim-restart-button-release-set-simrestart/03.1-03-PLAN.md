class SimChild { public: virtual const char *direction() const = 0; virtual void on_value(float v) {}  // line 33: -Wunused-parameter under -Wextra -Werror
                 virtual void on_connected() {} virtual void on_disconnected() {} void set_channel(const std::string &c); ... };
class SimBus : public Component { public:
  static constexpr size_t kMaxTx = 65536; static constexpr uint32_t kBackoffMin = 100; static constexpr uint32_t kBackoffMax = 2000;
  void set_path(const std::string &p); void register_child(SimChild *c); void setup() override; void loop() override;
  bool connected() const;  // fd_ >= 0
  void set_out(const std::string &ch, float v);   // dedupes unchanged values; sends + flush_() when connected
 protected:
  void try_connect_(); void close_(); void flush_(); void bump_backoff_();
  int fd_{-1}; uint32_t last_try_{0}; uint32_t retry_delay_{0}; uint32_t backoff_{kBackoffMin}; bool welcomed_{false};
  std::string rx_, tx_; std::vector<std::pair<std::string, float>> last_out_; };
}
// sim_bus.cpp behaviour the test pins:
//   flush_(): send(MSG_NOSIGNAL|MSG_DONTWAIT) until EAGAIN; then if tx_.size() > kMaxTx:
//     ESP_LOGW(TAG, "sim bus: %u bytes pending for a simulator that does not read; dropping the connection", ...); close_()
//   close_(): unregister+close fd, clear rx_/tx_, last_try_ = millis(); welcomed -> retry_delay_ = backoff_, else bump_backoff_()

//   try_connect_(): queues hello (k hello, proto 1, chans, t, name), one chan per child, last_out_ values as out frames, sync; children on_connected(); flush_()
//   a welcome line {"k":"welcome","proto":1,"t":0} read in loop() sets welcomed_ and backoff_ = kBackoffMin
